package "Controller" {

  class AuthController <<controller>> {
    + registerCustomer(data): Customer
    + login(email, password): Boolean
    + logout(userID): void
  }

  class ConcertController <<controller>> {
    + searchConcert(keyword): List<Concert>
    + getConcertList(): List<Concert>
    + getConcertDetail(concertID): Concert
    + createConcert(details): Concert
    + updateConcert(concertID, details): void
  }

  class BookingController <<controller>> {
    + createBooking(customerID, concertID, seats): Booking
    + getBookingHistory(customerID): List<Booking>
    + getCustomerContact(concertID): List<Customer>
  }

  class PaymentController <<controller>> {
    + processPayment(bookingID, method): Boolean
    + verifyPayment(paymentID): Boolean
  }

  class TicketController <<controller>> {
    + getTickets(customerID): List<Ticket>
  }

  class UserController <<controller>> {
    + getProfile(userID): User
    + updateProfile(userID, data): void
    + changePassword(userID, password): void
  }

  class SeatMapController <<controller>> {
    + createZone(concertID, details): Zone
    + updateZone(zoneID, details): void
    + defineSeatMap(zoneID, seats): void
    + updateSeat(seatID, details): void
  }

  class ReportController <<controller>> {
    + getMonthlySales(concertID): Report
    + getYearlySales(concertID): Report
  }

  class StaffController <<controller>> {
    + createStaff(companyID, data): Staff
  }

  class TicketValidationController <<controller>> {
    + scanTicketQR(qrCode): Boolean
    + validateTicketNumber(ticketID): Boolean
    + getValidationHistory(concertID): List<TicketValidationLog>
  }

  class UserManagementController <<controller>> {
    + getUsers(): List<User>
    + searchUser(keyword): List<User>
    + banUser(userID): void
    + unbanUser(userID): void
    + changeOrganizerCompany(userID, companyID): void
    + createOrganizer(data): EventOrganizer
    + createAdmin(data): Admin
  }

  class BookingManagementController <<controller>> {
    + getBookings(): List<Booking>
    + searchBooking(keyword): List<Booking>
  }

  class AuditLogController <<controller>> {
    + getLogs(): List<AuditLog>
    + searchLogs(keyword): List<AuditLog>
  }

  class AnomalyController <<controller>> {
    + getAnomalies(): List<SystemAnomaly>
    + getAnomalyDetail(anomalyID): SystemAnomaly
    + updateAnomalyStatus(anomalyID, status): void
    + executeAction(anomalyID, action): void
  }
}


' =========================================================
' ENTITY
' =========================================================

package "Entity" {

  class Company <<entity>> {
    - companyID: String
    - name: String
    - initials: String
    - description: String

    + getCompanyInfo(): String
    + listEvents(): List<Concert>
  }


  abstract class User <<entity>> {
    - userID: String
    - name: String
    - email: String
    - phone: String
    - password: Hash
    - role: Role
    - isBanned: Boolean

    + login(): Boolean
    + logout(): void
  }


  class Customer <<entity>> {
    + register(): void
    + searchConcert(keyword): List<Concert>
    + createBooking(concert, seats): Booking
    + viewHistory(): List<Booking>
    + viewTickets(): List<Ticket>
  }


  class EventOrganizer <<entity>> {
    - companyID: String

    + createConcert(details): Concert
    + defineSeatMap(concert, mapDetails): void
    + viewSalesRealTime(concert): Report
  }


  class Staff <<entity>> {
    - companyID: String

    + scanTicketQR(qrCode): Boolean
  }


  class Admin <<entity>> {
    + manageUser(userID, action): void
    + viewAnomalies(filters): List<SystemAnomaly>
    + acknowledgeAnomaly(anomalyID): void
    + resolveAnomaly(anomalyID, reason, files): void
    + executeQuickAction(anomalyID, actionType): void
  }


  class Concert <<entity>> {
    - concertID: String
    - name: String
    - artist: String
    - dateTime: DateTime
    - venue: String

    + getDetails(): ConcertInfo
  }


  class Zone <<entity>> {
    - zoneID: String
    - name: String
    - price: Double
    - capacity: Integer

    + getAvailableSeats(): List<Seat>
  }


  class Seat <<entity>> {
    - seatID: String
    - row: String
    - number: Integer
    - status: SeatStatus

    + lockSeat(): void
    + unlockSeat(): void
    + forceUnlock(): void
    + confirmSeat(): void
  }


  class Booking <<entity>> {
    - bookingID: String
    - bookingDate: DateTime
    - totalAmount: Double
    - status: BookingStatus
    - expiryTime: DateTime

    + calculateTotal(): Double
    + confirmBooking(): void
    + cancelBooking(): void
    + checkExpiry(): Boolean
  }


  class Payment <<entity>> {
    - paymentID: String
    - amount: Double
    - method: PaymentMethod
    - status: PaymentStatus
    - paymentDate: DateTime

    + processPayment(): Boolean
    + verifyStatus(): Boolean
  }


  class Ticket <<entity>> {
    - ticketID: String
    - qrCode: String
    - isUsed: Boolean

    + generateQR(): String
    + validateTicket(): Boolean
    + markAsUsed(): void
  }


  class SystemAnomaly <<entity>> {
    - anomalyID: String
    - type: AnomalyType
    - message: String
    - timestamp: DateTime
    - status: AnomalyStatus
    - resolveReason: String
    - attachments: List<String>

    + updateStatus(newStatus): void
    + addAttachment(file): void
  }


  class AuditLog <<entity>> {
    - logID: String
    - userID: String
    - action: String
    - timestamp: DateTime
    - ipAddress: String

    + getDetails(): String
  }


  class TicketValidationLog <<entity>> {
    - validationID: String
    - ticketID: String
    - staffID: String
    - validationTime: DateTime
    - result: Boolean
  }
}

User <|-- Customer

User <|-- Admin


' =========================================================
' ENTITY RELATIONSHIPS
' =========================================================

Company "1" *-- "*" EventOrganizer : employs >
Company "1" *-- "*" Staff : employs >
Company "1" -- "*" Concert : owns/hosts >

EventOrganizer "1" -- "*" Concert : manages >

Concert "1" *-- "*" Zone : contains >
Zone "1" *-- "*" Seat : contains >

Customer "1" -- "*" Booking : makes >
Booking "1" -- "*" Seat : reserves >
Booking "1" -- "1" Payment : has >
Booking "1" -- "*" Ticket : generates >

Ticket "1" -- "1" Seat : assigned to >
Ticket "*" -- "1" Concert : belongs to >

Staff "*" -- "*" Concert : scans tickets for >

TicketValidationLog "*" -- "1" Ticket : validates >
TicketValidationLog "*" -- "1" Staff : performed by >

SystemAnomaly "*" -- "1" Booking : references >
Admin "1" -- "*" SystemAnomaly : reviews >

AuditLog "*" -- "1" User : performed by >
