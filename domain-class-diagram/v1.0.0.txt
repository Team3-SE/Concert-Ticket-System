' Enums for Status
enum Role {
  CUSTOMER
  ORGANIZER
  ADMIN
}

enum SeatStatus {
  AVAILABLE
  LOCKED
  BOOKED
}

enum BookingStatus {
  PENDING
  COMPLETED
  CANCELLED
}

enum PaymentMethod {
  PROMPTPAY
  CREDIT_CARD
}

' Users
abstract class User {
  - userID: String
  - name: String
  - email: String
  - phone: String
  - password: Hash
  - role: Role
  + login(): Boolean
  + logout(): void
  + updateProfile(): void
}

class Customer {
  + register(): void
  + searchConcert(keyword): List<Concert>
  + createBooking(concert, seats): Booking
  + viewHistory(): List<Booking>
  + viewTickets(): List<Ticket>
}

class EventOrganizer {
  + createConcert(details): Concert
  + defineSeatMap(concert, mapDetails): void
  + viewSalesRealTime(concert): Report
  + generateReport(concert): Report
  + scanTicketQR(qrCode): Boolean
}

class Admin {
  + manageUser(userID, action): void
  + viewSystemLogs(): List<Log>
  + resolveIssue(issueID): void
}

User <|-- Customer
User <|-- EventOrganizer
User <|-- Admin

' Core System
class Concert {
  - concertID: String
  - name: String
  - artist: String
  - dateTime: DateTime
  - venue: String
  - description: String
  + getDetails(): ConcertInfo
  + checkAvailability(): Boolean
}

class Zone {
  - zoneID: String
  - name: String
  - price: Double
  - capacity: Integer
  + getAvailableSeats(): List<Seat>
}

class Seat {
  - seatID: String
  - row: String
  - number: Integer
  - status: SeatStatus
  + lockSeat(): void
  + unlockSeat(): void
  + confirmSeat(): void
}

class Booking {
  - bookingID: String
  - bookingDate: DateTime
  - totalAmount: Double
  - status: BookingStatus
  - expiryTime: DateTime
  + calculateTotal(): Double
  + confirmBooking(): void
  + cancelBooking(): void
  + checkExpiry(): Boolean
}

class Payment {
  - paymentID: String
  - amount: Double
  - method: PaymentMethod
  - paymentDate: DateTime
  - isSuccess: Boolean
  + processPayment(): Boolean
  + verifyStatus(): Boolean
}

class Ticket {
  - ticketID: String
  - qrCode: String
  - isUsed: Boolean
  + generateQR(): String
  + validateTicket(): Boolean
}

' Relationships
EventOrganizer "1" -- "*" Concert : manages >
Concert "1" *-- "*" Zone : contains >
Zone "1" *-- "*" Seat : contains >

Customer "1" -- "*" Booking : makes >
Booking "1" -- "*" Seat : reserves >
Booking "1" -- "1" Payment : has >
Booking "1" -- "*" Ticket : generates >

Ticket "1" -- "1" Seat : assigned to >
Ticket "*" -- "1" Concert : belongs to >
