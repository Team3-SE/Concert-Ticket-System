    %% ==============================
    %% APP_USER INHERITANCE
    %% ==============================

    APP_USER --o CUSTOMER  is
    APP_USER --o EVENT_ORGANIZER  is
    APP_USER --o STAFF  is
    APP_USER --o ADMIN  is


    %% ==============================
    %% ORGANIZATION
    %% ==============================

    ORGANIZATION --o{ EVENT_ORGANIZER  employs
    ORGANIZATION --o{ STAFF  employs
    ORGANIZATION --o{ CONCERT  owns

    EVENT_ORGANIZER --o{ STAFF  creates
    EVENT_ORGANIZER --o{ CONCERT  manages


    %% ==============================
    %% CONCERT
    %% ==============================

    CONCERT --o{ CONCERT_CONTACT  has
    CONCERT --o{ ZONE  contains

    ZONE --o ZONE_SEAT_PATTERN  has
    ZONE --o{ SEAT  contains


    %% ==============================
    %% BOOKING
    %% ==============================

    CUSTOMER --o{ BOOKING  makes
    CONCERT --o{ BOOKING  receives

    BOOKING --o{ BOOKING_SEAT  contains
    SEAT --o{ BOOKING_SEAT  reserved_in

    BOOKING --o{ BOOKING_STANDING_ITEM  contains
    ZONE --o{ BOOKING_STANDING_ITEM  reserved_in

    BOOKING --o PAYMENT  has
    BOOKING --o CUSTOMER_ORDER  generates
    PAYMENT --o CUSTOMER_ORDER  generates
    PAYMENT --o{ CHARGEBACK  has


    %% ==============================
    %% TICKET
    %% ==============================

    CUSTOMER_ORDER --o{ TICKET  generates
    CONCERT --o{ TICKET  has
    ZONE --o{ TICKET  has
    SEAT --o TICKET  assigned_to


    %% ==============================
    %% SEAT LOCK
    %% ==============================

    BOOKING --o{ SEAT_LOCK_LOG  has
    SEAT --o{ SEAT_LOCK_LOG  has


    %% ==============================
    %% STAFF  TICKET VALIDATION
    %% ==============================

    STAFF --o{ STAFF_CONCERT  assigned
    CONCERT --o{ STAFF_CONCERT  has_staff

    STAFF --o{ TICKET_VALIDATION_LOG  performs
    TICKET --o{ TICKET_VALIDATION_LOG  validated


    %% ==============================
    %% SYSTEM ANOMALY
    %% ==============================

    BOOKING --o{ SYSTEM_ANOMALY  has
    ADMIN --o{ SYSTEM_ANOMALY  reviews
    SYSTEM_ANOMALY --o{ ANOMALY_ATTACHMENT  has


    %% ==============================
    %% AUDIT LOG
    %% ==============================

    APP_USER --o{ AUDIT_LOG  generates
