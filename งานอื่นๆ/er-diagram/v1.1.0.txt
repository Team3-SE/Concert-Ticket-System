    COMPANY {

        string description
    }

    USER {
        string user_id PK
        string name
        string email UK
        string phone
        string password_hash
        string role
        boolean is_banned
    }

    CUSTOMER {
        string customer_id PK,FK
    }

    EVENT_ORGANIZER {
        string organizer_id PK,FK
        string company_id FK
    }

    STAFF {
        string staff_id PK,FK
        string company_id FK
    }

    ADMIN {
        string admin_id PK,FK
    }

    CONCERT {
        string concert_id PK
        string company_id FK
        string organizer_id FK
        string name
        string artist
        datetime date_time
        string venue
    }

    ZONE {
        string zone_id PK
        string concert_id FK
        string name
        decimal price
        int capacity
    }

    SEAT {
        string seat_id PK
        string zone_id FK
        string row
        int number
        string status
    }

    BOOKING {
        string booking_id PK
        string customer_id FK
        string concert_id FK
        datetime booking_date
        decimal total_amount
        string status
        datetime expiry_time
    }

    BOOKING_SEAT {
        string booking_id PK,FK
        string seat_id PK,FK
    }

    PAYMENT {
        string payment_id PK
        string booking_id FK,UK
        decimal amount
        string method
        string status
        datetime payment_date
        string provider_transaction_id
        string decline_code
        string risk_level
        string card_fingerprint
    }

    TICKET {
        string ticket_id PK
        string booking_id FK
        string concert_id FK
        string seat_id FK,UK
        string qr_code UK
        boolean is_used
    }

    SEAT_LOCK_LOG {
        string lock_id PK
        string booking_id FK
        string seat_id FK
        datetime locked_at
        datetime expired_at
        datetime released_at
        string status
    }

    CHARGEBACK {
        string chargeback_id PK
        string payment_id FK
        string provider_chargeback_id UK
        string reason
        decimal amount
        string status
        datetime created_at
        datetime resolved_at
    }

    STAFF_CONCERT {
        string staff_id PK,FK
        string concert_id PK,FK
    }

    TICKET_VALIDATION_LOG {
        string validation_id PK
        string ticket_id FK
        string staff_id FK
        datetime validation_time
        boolean result
    }

    SYSTEM_ANOMALY {
        string anomaly_id PK
        string booking_id FK
        string reviewed_by_admin_id FK
        string type
        string message
        datetime timestamp
        string status
        string resolve_reason
    }

    ANOMALY_ATTACHMENT {
        string attachment_id PK
        string anomaly_id FK
        string file_path
    }

    AUDIT_LOG {
        string log_id PK
        string user_id FK
        string action
        datetime timestamp
        string ip_address
    }


    %% ==============================
    %% USER INHERITANCE
    %% ==============================

    USER ||--o| CUSTOMER : "is"
    USER ||--o| EVENT_ORGANIZER : "is"
    USER ||--o| STAFF : "is"
    USER ||--o| ADMIN : "is"


    %% ==============================
    %% COMPANY / CONCERT
    %% ==============================

    COMPANY ||--o{ EVENT_ORGANIZER : employs
    COMPANY ||--o{ STAFF : employs
    COMPANY ||--o{ CONCERT : organizes

    EVENT_ORGANIZER ||--o{ CONCERT : manages

    CONCERT ||--|{ ZONE : contains
    ZONE ||--|{ SEAT : contains


    %% ==============================
    %% BOOKING
    %% ==============================

    CUSTOMER ||--o{ BOOKING : makes
    CONCERT ||--o{ BOOKING : receives

    BOOKING ||--o{ BOOKING_SEAT : contains
    SEAT ||--o{ BOOKING_SEAT : reserved_in

    BOOKING ||--o| PAYMENT : has
    BOOKING ||--o{ TICKET : generates

    BOOKING ||--o{ SEAT_LOCK_LOG : records
    SEAT ||--o{ SEAT_LOCK_LOG : has_lock_history


    %% ==============================
    %% PAYMENT / CHARGEBACK
    %% ==============================

    PAYMENT ||--o{ CHARGEBACK : may_have


    %% ==============================
    %% TICKET
    %% ==============================

    CONCERT ||--o{ TICKET : has
    SEAT ||--o| TICKET : assigned_to


    %% ==============================
    %% STAFF / CONCERT / VALIDATION
    %% ==============================

    STAFF ||--o{ STAFF_CONCERT : assigned
    CONCERT ||--o{ STAFF_CONCERT : has_staff

    STAFF ||--o{ TICKET_VALIDATION_LOG : performs
    TICKET ||--o{ TICKET_VALIDATION_LOG : validated


    %% ==============================
    %% SYSTEM ANOMALY
    %% ==============================

    BOOKING ||--o{ SYSTEM_ANOMALY : may_trigger
    ADMIN ||--o{ SYSTEM_ANOMALY : reviews

    SYSTEM_ANOMALY ||--o{ ANOMALY_ATTACHMENT : has
