sequenceDiagram
    autonumber

    actor Admin as Admin
    participant AuditUI@{ "type": "boundary" } as :AuditLogInterface
    participant AuditCtrl@{ "type": "control" } as :AuditLogController
    participant LogEntity@{ "type": "entity" } as :AuditLog

    %% 1. ดูรายการและค้นหา Log
    rect rgb(255, 248, 240)
    note right of Admin: 1. ดูรายการและค้นหา Log (View & Search Logs)
    Admin->>AuditUI: displayLogs() / searchLogs(keyword)
    activate AuditUI
    
    alt ดู Log ทั้งหมด
        AuditUI->>AuditCtrl: getLogs()
        activate AuditCtrl
        AuditCtrl->>LogEntity: ดึงข้อมูล Audit Log
        activate LogEntity
        LogEntity->>LogEntity: getDetails()
        LogEntity-->>AuditCtrl: List<AuditLog>
        deactivate LogEntity
        AuditCtrl-->>AuditUI: List<AuditLog>
        deactivate AuditCtrl
    else ค้นหา Log (ตาม IP, User, Action)
        AuditUI->>AuditCtrl: searchLogs(keyword)
        activate AuditCtrl
        AuditCtrl->>LogEntity: ค้นหาข้อมูลตาม keyword
        activate LogEntity
        LogEntity->>LogEntity: getDetails()
        LogEntity-->>AuditCtrl: List<AuditLog>
        deactivate LogEntity
        AuditCtrl-->>AuditUI: List<AuditLog>
        deactivate AuditCtrl
    end
    
    AuditUI-->>Admin: แสดงรายการ Audit Log
    deactivate AuditUI
    end
